class InviteAdminController(private val inviteService: InviteService, @Value("\${app.invite.admin-token:}") private val adminToken: String) {
    @PostMapping fun issue(@RequestHeader(value = "X-Admin-Token", required = false) token: String?, @Valid @RequestBody body: IssueInviteRequest): ResponseEntity<IssueInviteResponse> {
        requireValidAdminToken(token)
        return ResponseEntity.status(HttpStatus.CREATED).body(IssueInviteResponse(inviteService.create(body.referrerUserId)))
    }
    private fun requireValidAdminToken(provided: String?) {
        if (adminToken.isBlank() || provided == null || !MessageDigest.isEqual(provided.toByteArray(Charsets.UTF_8), adminToken.toByteArray(Charsets.UTF_8))) throw AdminAuthException()
    }
}
